class Definition(StrictModel):
    base_images:dict[str,str]
    files:dict[str,str]
    dockerfiles:dict[str,str]
    generated_source_sha256:str
    training_validation_digest:str
    source_digest:str
class Result(StrictModel):
    definition_digest:str
    request_id:str
    operation_id:str
    images:dict[str,str]
    files:dict[str,str]
async def main():
    assert verification_source_digest(ROOT)==SOURCE
    r=Router.from_manifest(ROOT/'.aeep/live-review-v3/workbook-phase-final.json')
    try:
        s=AssessmentService(r,ROOT/'.aeep/live-review-v3/.aeep/assessments');repo=s.repository
        built=ReusableBuildOutput.model_validate(repo.get('reusable_build_output','reusable-build:planning_bd963159f568438aa360dbab05e72075'))
        validation=json.loads((P/'reusable-phase-final-training-result.json').read_text());assert len(validation['results'])==28 and all(x['valid'] for x in validation['results'])
        sha=hashlib.sha256(built.source.encode()).hexdigest();assert sha==validation['generated_source_sha256']
        usage=built.usage+'\n'
        files={'/opt/dependencies/reusable/clean_workbook.py':sha,'/opt/dependencies/reusable/USAGE.txt':hashlib.sha256(usage.encode()).hexdigest()}
        tags={role:f'aeep-reusable-reviewed-base-{role}:{sha[:16]}' for role in BASES}
        dockerfiles={role:f'FROM {tags[role]}\nCOPY --chmod=0444 clean_workbook.py USAGE.txt /opt/dependencies/reusable/\n' for role in BASES}
        definition=Definition(base_images=BASES,files=files,dockerfiles=dockerfiles,generated_source_sha256=sha,training_validation_digest=content_digest(validation),source_digest=SOURCE)
        digest=repo.put('reusable_image_definition',content_digest(definition),definition)
        prior=RecipeMaterializationRequest.model_validate(repo.get('recipe_materialization_request',validation['request_id']))
        request=prior.model_copy(update={'plan_id':new_id('recipe_generation'),'definition_digests':[*prior.definition_digests,digest]})
        repo.put('recipe_materialization_request',request.plan_id,request)
        ds=set(request.definition_digests)|{request.subject_digest}
        definitions={d:json.loads(r.store._connection.execute('SELECT payload_json FROM assessment_records WHERE digest=?',(d,)).fetchone()[0]) for d in ds}
        root=AssessmentAuthorization.model_validate(repo.get('authorization','onboarding'))
        amendment=AssessmentScopeAmendment(authorization_id='onboarding',authorization_digest=content_digest(root),subject_digests=[request.subject_digest],recipe_digests=[request.recipe_digest],environment_digests=[request.environment_digest],reviewed_digests=sorted(ds))
        review={'authority':'2026-09-27 delegated routine definition and testing authority','purpose':'Freeze unchanged generated baseline after 28/28 contained training validation. Two derivative images allow the existing paired boundary verifier to check its shared presence; only control derivative enters the reusable arm. No trial model, credential mount, candidate execution or network access.','definition':definition.model_dump(mode='json'),'request':request.model_dump(mode='json'),'amendment':amendment.model_dump(mode='json'),'definitions':definitions,'maximum_operations':1,'maximum_elapsed_seconds':600,'maximum_model_turns':0}
        with (P/'reusable-phase-final-image-review.json').open('x') as f:json.dump(review,f,indent=2);f.write('\n')
        repo.approve_bundle(amendment,definitions)
        operation=f'{request.plan_id}-image-setup';repo.reserve(request,operation,AssessmentLimits(max_operations=1,max_elapsed_seconds=600),stage='reusable_image_setup')
        started=time.perf_counter();images={}
        try:
            with tempfile.TemporaryDirectory(prefix='aeep-reusable-frozen-') as name:
                context=Path(name);(context/'clean_workbook.py').write_text(built.source);(context/'USAGE.txt').write_text(usage)
                for role,base in BASES.items():
                    repo.authorize(request);assert verification_source_digest(ROOT)==SOURCE
                    inspected=json.loads(subprocess.check_output([*DOCKER,'image','inspect',base],text=True))[0];assert inspected['Id']==base
                    subprocess.run([*DOCKER,'tag',base,tags[role]],check=True)
                    assert json.loads(subprocess.check_output([*DOCKER,'image','inspect',tags[role]],text=True))[0]['Id']==base
                    (context/'Dockerfile').write_text(dockerfiles[role])
                    with (P/f'reusable-phase-final-image-{role}-build.txt').open('x') as log:
                        subprocess.run([*DOCKER,'build','--pull=false','--network=none','--iidfile',str(context/'image.id'),str(context)],check=True,stdout=log,stderr=subprocess.STDOUT,timeout=240)
                    image=(context/'image.id').read_text().strip();derived=json.loads(subprocess.check_output([*DOCKER,'image','inspect',image],text=True))[0]
                    assert derived['RootFS']['Layers'][:len(inspected['RootFS']['Layers'])]==inspected['RootFS']['Layers']
                    assert json.loads(subprocess.check_output([*DOCKER,'image','inspect',tags[role]],text=True))[0]['Id']==base
                    check='import hashlib,json,pathlib;print(json.dumps({p:hashlib.sha256(pathlib.Path(p).read_bytes()).hexdigest() for p in '+repr(list(files))+'}))'
                    actual=json.loads(subprocess.check_output([*DOCKER,'run','--rm','--network=none','--read-only','--user','65534:65534','--cap-drop=ALL','--security-opt=no-new-privileges','--cpus','1','--memory','256m','--pids-limit','32','--entrypoint','python3',image,'-c',check],text=True,timeout=20))

        finally:repo.finish_operation(operation,elapsed_seconds=time.perf_counter()-started)
